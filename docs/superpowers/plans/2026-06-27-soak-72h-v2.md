## Task 2: Write Failing Regression Test for multi-VALUES INSERT

- [ ] **Step 2.1: Create test file with failing case**

```rust
// crates/executor/tests/bulk_insert_regression_test.rs
// Regression test for Issue #3575: multi-VALUES INSERT must not lose connection.

use sqlrustgo::{ExecutionEngine, MemoryStorage, StorageEngine};
use sqlrustgo_types::Value as SqlValue;
use std::sync::Arc;
use std::time::Instant;

#[test]
fn test_bulk_insert_5000_rows_does_not_disconnect() {
    let mut engine = ExecutionEngine::new(Arc::new(std::sync::RwLock::new(MemoryStorage::new())));

    engine.execute("CREATE TABLE sbtest1 (id INTEGER PRIMARY KEY, k INTEGER, c VARCHAR(120), pad VARCHAR(60))")
        .expect("CREATE TABLE");

    // Build a single INSERT with 5000 rows (sysbench bulk_insert pattern)
    let mut sql = String::from("INSERT INTO sbtest1 VALUES ");
    for i in 0..5000 {
        if i > 0 { sql.push(','); }
        sql.push_str(&format!("({}, {}, 'c_{}', 'p_{}')", i, i % 1000, i, i));
    }

    let start = Instant::now();
    let result = engine.execute(&sql);
    let elapsed = start.elapsed();

    assert!(result.is_ok(), "bulk_insert failed: {:?}", result.err());
    assert!(elapsed.as_secs() < 30, "bulk_insert too slow: {:?}", elapsed);

    // Verify count
    let count_result = engine.execute("SELECT COUNT(*) FROM sbtest1")
        .expect("SELECT COUNT");
    assert!(count_result.rows.len() > 0);
    let count: i64 = match &count_result.rows[0][0] {
        SqlValue::Integer(n) => *n,
        _ => panic!("unexpected type"),
    };
    assert_eq!(count, 5000, "expected 5000 rows, got {}", count);
}
```

## Task 5: Write Failing Test for mysql CLI TLS

**Files:**
- Create: `crates/mysql-server/tests/mysql_cli_tls_handshake_test.rs`

- [ ] **Step 5.1: Create test simulating mysql CLI handshake**

```rust
// crates/mysql-server/tests/mysql_cli_tls_handshake_test.rs
// Regression test: server must accept mysql CLI clients (no TLS upgrade request).

use std::net::TcpStream;
use std::io::{Read, Write};

#[test]
fn test_mysql_cli_handshake_no_tls_upgrade() {
    // Start server in test mode (--auth-mode=none, port 13397)
    // (Or use existing test harness)
    
    // Connect via raw TCP (simulating mysql CLI default)
    let mut stream = TcpStream::connect("127.0.0.1:13397")
        .expect("connect");
    
    // Read server handshake packet (greeting)
    let mut buf = [0u8; 256];
    let n = stream.read(&mut buf).expect("read handshake");
    assert!(n > 4, "handshake too short");
    
    // Check capability flags byte at offset 5 (lower 8 bits)
    // bit 9 (0x200) = CLIENT_SSL. If set, server expects TLS upgrade.
    // mysql CLI by default does NOT send TLS upgrade.
    // 
    // Assertion: handshake should NOT advertise CLIENT_SSL,
    // OR server should accept non-TLS clients even if it advertised.
    let capability_lower = buf[5];
    let client_ssl_bit = 0x08u8; // bit 9 in the second capability byte (offset 5)
    
    // Expected: capability does not have CLIENT_SSL forced.
    // If we advertise TLS, we must support fallback.
    if capability_lower & client_ssl_bit != 0 {
        // OK if we advertised, but must also support non-TLS path.
        // For now, log; the fix is in Task 6.
        eprintln!("WARN: server advertised CLIENT_SSL in handshake");
    }
    
    // Send minimal handshake response (no TLS upgrade)
    let auth_response = b"\x85\xaa\xbf\x19\x00\x00\x00\x01\xff\x00\x00\x00\x00\x00\x00\x00\
                          \x72\x6f\x6f\x74\x00\x00\x6d\x79\x73\x71\x6c\x5f\x6e\x61\x74\x69\
                          \x76\x65\x5f\x70\x61\x73\x73\x77\x6f\x72\x64\x00";
    stream.write_all(auth_response).expect("write auth");
    
    // Read response (OK or ERR)
    let mut resp = [0u8; 64];
    let rn = stream.read(&mut resp).expect("read response");
    assert!(rn > 0, "no response");
    
    // Response first byte: 0x00 = OK, 0xff = ERR
    let status = resp[4]; // packet header (4 bytes) + payload first byte
    assert_eq!(status, 0x00, "expected OK (0x00), got 0x{:02x}", status);
}
```
